def connect_db():
    """Create a SQLite connection with row access and foreign keys."""
    DB_PATH.parent.mkdir(parents=True, exist_ok=True)

    conn = sqlite3.connect(DB_PATH, timeout=10)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA foreign_keys = ON")

    return conn


def utc_now():
    """Return the current UTC time as an ISO 8601 string."""
    return datetime.now(timezone.utc).isoformat()


def hash_key(raw_key: str) -> str:
    """Hash a high-entropy API key before storing or looking it up."""
    return hashlib.sha256(raw_key.encode("utf-8")).hexdigest()


def _ensure_column(
    conn,
    table: str,
    column: str,
    definition: str,
):
    """Add a column only when it does not already exist."""
    columns = {
        row["name"]
        for row in conn.execute(
            f"PRAGMA table_info({table})"
        ).fetchall()
    }

    if column not in columns:
        conn.execute(
            f"ALTER TABLE {table} ADD COLUMN {column} {definition}"
        )

def init_db():
    """
    Initialize tables and apply additive migrations.

    Existing database records are preserved.
    """
    with connect_db() as conn:
        conn.executescript(
            """
            CREATE TABLE IF NOT EXISTS users (
                id TEXT PRIMARY KEY,
                email TEXT NOT NULL UNIQUE,
                role TEXT NOT NULL
                    CHECK (role IN ('admin', 'member')),
                created_at TEXT NOT NULL
            );

            CREATE TABLE IF NOT EXISTS api_keys (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id TEXT NOT NULL,
                key_hash TEXT NOT NULL UNIQUE,
                revoked INTEGER NOT NULL DEFAULT 0,
                created_at TEXT NOT NULL,
                FOREIGN KEY (user_id) REFERENCES users(id)
            );

            CREATE TABLE IF NOT EXISTS audit_logs (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                user_id TEXT NOT NULL,
                server_name TEXT NOT NULL,
                tool_name TEXT NOT NULL,
                timestamp TEXT NOT NULL,
                duration_ms INTEGER NOT NULL,
                outcome TEXT NOT NULL,
                error TEXT
            );

            CREATE TABLE IF NOT EXISTS upstreams (
                name TEXT PRIMARY KEY,
                url TEXT NOT NULL,
                enabled INTEGER NOT NULL DEFAULT 1,
                created_at TEXT NOT NULL
            );

        )

        # User migrations
        _ensure_column(conn, "users", "username", "TEXT")

def _upstream_to_dict(row, include_secrets: bool = False):
    """
    Convert an upstream database row into a dictionary.

    Ciphertext and nonce are never returned. Plaintext credentials
    are included only when explicitly requested by internal code.
    """
    result = {
        "name": row["name"],
        "url": row["url"],
        "enabled": bool(row["enabled"]),
        "created_at": row["created_at"],
        "auth_mode": row["auth_mode"],
        "custom_header_name": row["custom_header_name"],
        "timeout_seconds": row["timeout_seconds"],
        "connection_status": row["connection_status"],
        "consecutive_failures": row["consecutive_failures"],
        "last_checked_at": row["last_checked_at"],
        "last_success_at": row["last_success_at"],
        "last_discovery_at": row["last_discovery_at"],
        "last_error": row["last_error"],
        "circuit_open_until": row["circuit_open_until"],
        "hasCredential": bool(row["credential_ciphertext"]),
    }

    if include_secrets:
        ciphertext = row["credential_ciphertext"]
        nonce = row["credential_nonce"]

        result["credential"] = (
            decrypt_credential(ciphertext, nonce)
            if ciphertext and nonce
            else None
        )

    return result


def save_upstream(
    name: str,
    url: str,
    enabled: bool = True,
    auth_mode: str = "none",
    credential: str | None = None,
    custom_header_name: str | None = None,
    timeout_seconds: int = 15,
):
    """
    Create or update a persisted upstream configuration.

    Credentials are encrypted before they are written to SQLite.
    Existing credentials are preserved when no replacement is supplied.
    """
    name = name.strip()
    url = url.strip()
    auth_mode = auth_mode.strip().lower()

    if not name:
        raise ValueError("Upstream name is required")

    if not url:
        raise ValueError("Upstream URL is required")

    if auth_mode not in {"none", "bearer", "custom_header"}:
        raise ValueError(
            "auth_mode must be none, bearer, or custom_header"
        )

    if auth_mode == "custom_header":
        custom_header_name = (custom_header_name or "").strip()

        if not custom_header_name:
            raise ValueError(
                "custom_header_name is required for custom_header mode"
            )

        # Reject invalid HTTP header names and header injection.
        if (
            any(char in custom_header_name for char in "\r\n:")
            or not all(
                char.isalnum() or char in "!#$%&'*+-.^_`|~"
                for char in custom_header_name
            )
        ):
            raise ValueError("Invalid custom_header_name")
    else:
        custom_header_name = None

    try:
        timeout_seconds = int(timeout_seconds)
    except (TypeError, ValueError):
        raise ValueError(
            "timeout_seconds must be between 1 and 120"
        ) from None

    if not 1 <= timeout_seconds <= 120:
        raise ValueError("timeout_seconds must be between 1 and 120")

    with connect_db() as conn:
        existing = conn.execute(
            """
            SELECT *
            FROM upstreams
            WHERE name = ?
            """,
            (name,),
        ).fetchone()

        created_at = (
            existing["created_at"]
            if existing
            else utc_now()
        )

        old_url = existing["url"] if existing else None
        old_auth_mode = (
            existing["auth_mode"] if existing else "none"
        )
        old_header_name = (
            existing["custom_header_name"]
            if existing
            else None
        )

        # Keep the existing encrypted credential unless a replacement
        # is supplied or authentication is disabled.
        ciphertext = (
            existing["credential_ciphertext"]
            if existing
            else None
        )
        nonce = (
            existing["credential_nonce"]
            if existing
            else None
        )

        if auth_mode == "none":
            ciphertext = None
            nonce = None

        elif credential is not None:
            if credential.strip():
                ciphertext, nonce = encrypt_credential(credential)
            else:
                # An explicitly empty credential clears the secret.
                ciphertext = None
                nonce = None

        credential_changed = (
            credential is not None
            and credential.strip() != ""
        )

        configuration_changed = (
            existing is None
            or old_url != url
            or old_auth_mode != auth_mode
            or old_header_name != custom_header_name
            or credential_changed
        )

        if configuration_changed:
            connection_status = "unknown"
            consecutive_failures = 0
            last_checked_at = None
            last_success_at = None
            last_discovery_at = None
            last_error = None
            circuit_open_until = None
        else:
            connection_status = existing["connection_status"]
            consecutive_failures = existing["consecutive_failures"]
            last_checked_at = existing["last_checked_at"]
            last_success_at = existing["last_success_at"]
            last_discovery_at = existing["last_discovery_at"]
            last_error = existing["last_error"]
            circuit_open_until = existing["circuit_open_until"]

        conn.execute(
            """
            INSERT INTO upstreams (
                name,
                url,
                enabled,
                created_at,
                auth_mode,
                credential_ciphertext,
                credential_nonce,
                custom_header_name,
                timeout_seconds,
                connection_status,
                consecutive_failures,
                last_checked_at,
                last_success_at,
                last_discovery_at,
                last_error,
                circuit_open_until
            )
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
            ON CONFLICT(name) DO UPDATE SET
                url = excluded.url,
                enabled = excluded.enabled,
                auth_mode = excluded.auth_mode,
                credential_ciphertext = excluded.credential_ciphertext,
                credential_nonce = excluded.credential_nonce,
                custom_header_name = excluded.custom_header_name,
                timeout_seconds = excluded.timeout_seconds,
                connection_status = excluded.connection_status,
                consecutive_failures = excluded.consecutive_failures,
                last_checked_at = excluded.last_checked_at,
                last_success_at = excluded.last_success_at,
                last_discovery_at = excluded.last_discovery_at,
                last_error = excluded.last_error,
                circuit_open_until = excluded.circuit_open_until
            """,
            (
                name,
                url,
                int(enabled),
                created_at,
                auth_mode,
                ciphertext,
                nonce,
                custom_header_name,
                timeout_seconds,
                connection_status,
                consecutive_failures,
                last_checked_at,
                last_success_at,
                last_discovery_at,
                last_error,
                circuit_open_until,
            ),
        )

        row = conn.execute(
            """
            SELECT *
            FROM upstreams
            WHERE name = ?
            """,
            (name,),
        ).fetchone()

    # Never return ciphertext, nonce, or plaintext credentials here.
    return _upstream_to_dict(row)


def load_upstreams(include_secrets: bool = False):
    """
    Load persisted upstream configurations.

    By default, this returns metadata only. Set include_secrets=True
    only for trusted internal gateway code that needs credentials.
    """
    with connect_db() as conn:
        rows = conn.execute(
            """
            SELECT *
            FROM upstreams
            ORDER BY name
            """
        ).fetchall()

    return [
        _upstream_to_dict(
            row,
            include_secrets=include_secrets,
        )
        for row in rows
    ]


def delete_upstream_record(name: str):
    """Delete a persisted upstream configuration."""
    with connect_db() as conn:
        cursor = conn.execute(
            "DELETE FROM upstreams WHERE name = ?",
            (name.strip(),),
        )

    return cursor.rowcount > 0

def create_user(
    username: str,
    role: str = "member",
    email: str | None = None,
):
    """Create a user with the admin or member role."""
    username = username.strip()

    if not 2 <= len(username) <= 80:
        raise ValueError("Username must be between 2 and 80 characters")

    if role not in {"admin", "member"}:
        raise ValueError("Role must be admin or member")

    email = (email or f"{username}@local.invalid").strip().lower()

    if not email:
        raise ValueError("Email is required")

    user_id = secrets.token_hex(12)

    with connect_db() as conn:
        conn.execute(
            """
            INSERT INTO users (id, username, email, role, created_at)
            VALUES (?, ?, ?, ?, ?)
            """,
            (user_id, username, email, role, utc_now()),
        )

    return user_id


def issue_api_key(user_id: str, label: str = "default"):
    """
    Issue a new API key.

    The raw key is returned once. Only its SHA-256 hash is stored.
    """
    label = label.strip()

    if not label:
        raise ValueError("API key label is required")

    raw_key = "tmcp_" + secrets.token_urlsafe(32)

    with connect_db() as conn:
        user = conn.execute(
            "SELECT id FROM users WHERE id = ?",
            (user_id,),
        ).fetchone()

        if user is None:
            raise ValueError("User not found")

        cursor = conn.execute(
            """
            INSERT INTO api_keys (user_id, key_hash, label, key_prefix, created_at)
            VALUES (?, ?, ?, ?, ?)
            """,
            (user_id, hash_key(raw_key), label, raw_key[:10], utc_now()),
        )

        key_id = cursor.lastrowid

    return {
        "key_id": key_id,
        "user_id": user_id,
        "label": label,
        "api_key": raw_key,
        "warning": "Save this key now; it will not be shown again.",
    }


def authenticate_api_key(raw_key: str):
    """Return the user associated with a valid, non-revoked key."""
    if not raw_key:
        return None
    with connect_db() as conn:
        row = conn.execute(
            """
            SELECT
                u.id,
                u.username,
                u.email,
                u.role,
                k.id AS key_id
            FROM api_keys AS k
            JOIN users AS u ON u.id = k.user_id
            WHERE k.key_hash = ?
              AND k.revoked = 0
            """,
            (hash_key(raw_key),),
        ).fetchone()

        if row is None:
            return None

        cursor = conn.execute(
            """
            UPDATE api_keys
            SET last_used_at = ?
            WHERE id = ? AND revoked = 0
            """,
            (utc_now(), row["key_id"]),
        )

        if cursor.rowcount != 1:
            return None

        return dict(row)


def revoke_api_key(key_id: int):
    """Revoke an active API key."""
    with connect_db() as conn:
        cursor = conn.execute(
            """
            UPDATE api_keys
            SET revoked = 1, revoked_at = ?
            WHERE id = ? AND revoked = 0
            """,
            (utc_now(), key_id),
        )

    return cursor.rowcount > 0


def list_users():
    """List users with API-key usage summaries."""
    with connect_db() as conn:
        rows = conn.execute(
            """
            SELECT
                u.id AS user_id,
                u.username,
                u.email,
                u.role,
                u.created_at,
                COUNT(CASE WHEN k.revoked = 0 THEN 1 END) AS active_keys,
                COUNT(k.id) AS total_keys,
                MAX(k.last_used_at) AS last_used_at
            FROM users AS u
            LEFT JOIN api_keys AS k ON k.user_id = u.id
            GROUP BY u.id
            ORDER BY u.created_at ASC
            """
        ).fetchall()

    return [dict(row) for row in rows]


def _other_admin_count(conn, user_id: str) -> int:
    return conn.execute(
        "SELECT COUNT(*) FROM users WHERE role = 'admin' AND id != ?",
        (user_id,),
    ).fetchone()[0]


def update_user_role(user_id: str, role: str) -> bool:
    """Change a user's role. Refuses to demote the last admin."""
    if role not in {"admin", "member"}:
        raise ValueError("Role must be admin or member")

    with connect_db() as conn:
        row = conn.execute(
            "SELECT id, role FROM users WHERE id = ?", (user_id,)
        ).fetchone()

        if row is None:
            return False

        if (
            row["role"] == "admin"
            and role != "admin"
            and _other_admin_count(conn, user_id) == 0
        ):
            raise ValueError("Cannot demote the last admin")

        conn.execute(
            "UPDATE users SET role = ? WHERE id = ?", (role, user_id)
        )

    return True


def delete_user(user_id: str) -> bool:
    """Delete a user and their API keys. Audit history is kept."""
    with connect_db() as conn:
        row = conn.execute(
            "SELECT id, role FROM users WHERE id = ?", (user_id,)
        ).fetchone()

        if row is None:
            return False

        if row["role"] == "admin" and _other_admin_count(conn, user_id) == 0:
            raise ValueError("Cannot remove the last admin")

        conn.execute("DELETE FROM api_keys WHERE user_id = ?", (user_id,))
        conn.execute("DELETE FROM users WHERE id = ?", (user_id,))

    return True

def write_audit(
    user_id: str,
    username: str,
    server_name: str,
    tool_name: str,
    duration_ms: int,
    outcome: str,
    error_message: str | None = None,

    started_at = started_at or timestamp
    error_message = error_message[:500] if error_message else None

    with connect_db() as conn:
        conn.execute(
            """
            INSERT INTO audit_logs (
                user_id,
                username,
                server_name,
                tool_name,
                timestamp,
                started_at,
                duration_ms,
                outcome,
                error,
                error_message
            )

            """,
            (
                user_id,
                username,
                server_name,
                tool_name,
                timestamp,
                started_at,
                max(0, duration_ms),
                outcome,
                error_message,
                error_message,
            ),
        )

def extract_api_key(request: Request) -> str:
    """Accept API keys from x-api-key or Authorization: Bearer."""
    api_key = request.headers.get("x-api-key", "").strip()
    if api_key:
        return api_key
    authorization = request.headers.get("authorization", "").strip()
    scheme, separator, credential = authorization.partition(" ")
    if separator and scheme.lower() == "bearer":
        return credential.strip()
    return ""


class GatewayAuthMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        path = request.url.path

        # Kubernetes liveness/readiness check.
        if path == "/health":
            return await call_next(request)

        is_admin_path = (
            path == "/admin" or path.startswith("/admin/")
        )

        is_mcp_path = (
            path == "/mcp" or path.startswith("/mcp/")
        )

        # API-key-protected endpoints for logged-in users.
        is_member_api_path = (
            path == "/api" or path.startswith("/api/")
        )

        user = None

        if is_admin_path:
            # Bootstrap token supports initial administration.
            expected = os.getenv("ADMIN_BOOTSTRAP_TOKEN", "")
            supplied = request.headers.get("x-admin-token", "")

            if (
                expected
                and supplied
                and secrets.compare_digest(supplied, expected)
            ):
                user = {
                    "id": "bootstrap-admin",
                    "username": "bootstrap-admin",
                    "email": "bootstrap-admin",
                    "role": "admin",
                }
            else:
                # A registered admin may use an individual API key.
                user = authenticate_api_key(extract_api_key(request))

            if user is None:
                return JSONResponse(
                    {"detail": "Invalid admin credentials"},
                    status_code=401,
                )

            if user.get("role") != "admin":
                return JSONResponse(
                    {"detail": "Admin role required"},
                    status_code=403,
                )

        elif is_mcp_path or is_member_api_path:
            # MCP tools and member API endpoints require a valid key.
            user = authenticate_api_key(extract_api_key(request))

            if user is None:
                return JSONResponse(
                    {
                        "detail": (
                            "Missing, invalid, or revoked API key"
                        )
                    },
                    status_code=401,
                )

        else:
            # Unprotected routes, such as non-sensitive documentation.
            return await call_next(request)

        # Make the authenticated identity available to route handlers
        # and audit logging for the duration of this request.
        user_token = current_user_id.set(user["id"])
        role_token = current_user_role.set(user["role"])
        username_token = current_username.set(
            user.get("username") or user["email"]
        )

        request.state.user = user

        try:
            return await call_next(request)
        finally:
            current_user_id.reset(user_token)
            current_user_role.reset(role_token)
            current_username.reset(username_token)

import base64
import ipaddress
import os
import secrets
import socket
from urllib.parse import urlparse

from cryptography.hazmat.primitives.ciphers.aead import AESGCM


def _get_encryption_key() -> bytes:
    """Read a base64-encoded, 32-byte AES-256 key."""
    encoded = os.getenv("UPSTREAM_CREDENTIALS_KEY", "").strip()

    if not encoded:
        raise RuntimeError(
            "UPSTREAM_CREDENTIALS_KEY is not configured"
        )

    try:
        key = base64.urlsafe_b64decode(
            encoded + "=" * (-len(encoded) % 4)
        )
    except Exception as exc:
        raise RuntimeError(
            "UPSTREAM_CREDENTIALS_KEY must be valid URL-safe base64"
        ) from exc

    if len(key) != 32:
        raise RuntimeError(
            "UPSTREAM_CREDENTIALS_KEY must decode to exactly 32 bytes"
        )

    return key


def encrypt_credential(value: str) -> tuple[str, str]:
    """Return (ciphertext, nonce), both base64 encoded."""
    if not value:
        raise ValueError("Credential cannot be empty")

    nonce = secrets.token_bytes(12)
    aesgcm = AESGCM(_get_encryption_key())

    ciphertext = aesgcm.encrypt(
        nonce,
        value.encode("utf-8"),
        None,
    )

    return (
        base64.urlsafe_b64encode(ciphertext).decode("ascii"),
        base64.urlsafe_b64encode(nonce).decode("ascii"),
    )


def decrypt_credential(
    ciphertext: str | None,
    nonce: str | None,
) -> str | None:
    """Decrypt a stored credential or return None if absent."""
    if not ciphertext or not nonce:
        return None

    try:
        cipher_bytes = base64.urlsafe_b64decode(ciphertext)
        nonce_bytes = base64.urlsafe_b64decode(nonce)

        return AESGCM(_get_encryption_key()).decrypt(
            nonce_bytes,
            cipher_bytes,
            None,
        ).decode("utf-8")
    except Exception as exc:
        raise RuntimeError(
            "Could not decrypt upstream credential. "
            "Check the encryption key and database."
        ) from exc


def build_auth_headers(
    auth_mode: str,
    credential: str | None,
    custom_header_name: str | None = None,
) -> dict[str, str]:
    """Build request headers for one upstream."""
    mode = (auth_mode or "none").lower()

    if mode == "none":
        return {}

    if not credential:
        raise RuntimeError(
            f"Authentication mode '{mode}' requires a credential"
        )

    if mode == "bearer":
        return {"Authorization": f"Bearer {credential}"}

    if mode == "custom_header":
        header_name = (custom_header_name or "").strip()

        if not header_name:
            raise RuntimeError(
                "custom_header mode requires a header name"
            )

        if any(ch in header_name for ch in "\r\n:"):
            raise RuntimeError("Invalid custom header name")

        if any(ch in credential for ch in "\r\n"):
            raise RuntimeError("Invalid custom header credential")

        if header_name.lower() in {
            "host",
            "content-length",
            "transfer-encoding",
            "connection",
        }:
            raise RuntimeError(
                "This HTTP header cannot be used for upstream authentication"
            )

        return {header_name: credential}

    raise RuntimeError(
        f"Unsupported upstream authentication mode: {mode}"
    )


def validate_upstream_url(url: str) -> str:
    """
    Validate upstream URLs.

    UPSTREAM_ALLOWED_HOSTS, when configured, is a comma-separated
    hostname allowlist. Local/private hosts remain available for local
    development unless an allowlist is configured.
    """
    value = url.strip()
    parsed = urlparse(value)

    if parsed.scheme not in {"http", "https"}:
        raise ValueError("Upstream URL must use HTTP or HTTPS")

    if not parsed.hostname:
        raise ValueError("Upstream URL must contain a hostname")

    if parsed.username or parsed.password:
        raise ValueError(
            "Do not embed credentials in the upstream URL"
        )

    if parsed.fragment:
        raise ValueError("Upstream URL must not contain a fragment")

    hostname = parsed.hostname.rstrip(".").lower()

    if hostname in {
        "metadata.google.internal",
        "metadata",
    }:
        raise ValueError("This upstream hostname is not permitted")

    allowed_hosts = {
        item.strip().lower().rstrip(".")
        for item in os.getenv(
            "UPSTREAM_ALLOWED_HOSTS", ""
        ).split(",")
        if item.strip()
    }

    if allowed_hosts and hostname not in allowed_hosts:
        raise ValueError(
            "Upstream hostname is not in UPSTREAM_ALLOWED_HOSTS"
        )

    # Explicitly block common cloud metadata IPs.
    try:
        ip = ipaddress.ip_address(hostname)
        if ip.is_link_local:
            raise ValueError(
                "Link-local upstream addresses are not permitted"
            )
    except ValueError as exc:
        # Re-raise our link-local rejection, but allow ordinary DNS names.
        if "Link-local" in str(exc):
            raise

    return value.rstrip("/")
