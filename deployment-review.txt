def get_upstream_config(server_name: str) -> dict[str, Any]:
    config = UPSTREAM_CONFIG.get(server_name)
    if not config:
        raise RuntimeError(f"Unknown upstream server: {server_name}")
    if not config.get("enabled", False):
        raise RuntimeError(f"Upstream '{server_name}' is disabled")
    return config


def get_upstream_headers(config: dict[str, Any]) -> dict[str, str]:
    return build_auth_headers(
        auth_mode=config.get("auth_mode", "none"),
        credential=config.get("credential"),
        custom_header_name=config.get("custom_header_name"),
    )


def circuit_is_open(config: dict[str, Any]) -> bool:
    value = config.get("circuit_open_until")
    if not value:
        return False
    try:
        until = datetime.fromisoformat(value)
        if until.tzinfo is None:
            until = until.replace(tzinfo=timezone.utc)
        return datetime.now(timezone.utc) < until
    except (TypeError, ValueError):
        return False


def update_upstream_status(
    name: str,
    status: str,
    *,
    error: str | None = None,
    discovery_succeeded: bool = False,
):
    """Persist safe connection/health metadata without storing secrets."""
    now = datetime.now(timezone.utc).isoformat()
    config = UPSTREAM_CONFIG.get(name, {})
    failures = UPSTREAM_FAILURES.get(
        name, int(config.get("consecutive_failures") or 0)
    )

    if status == "connected":
        failures = 0
        circuit_open_until = None
        last_success_at = now
        last_error = None
    elif status == "disabled":
        failures = 0
        circuit_open_until = None
        last_success_at = config.get("last_success_at")
        last_error = None
    elif status == "circuit_open":
        circuit_open_until = (
            datetime.now(timezone.utc)
            + timedelta(seconds=UPSTREAM_CIRCUIT_OPEN_SECONDS)
        ).isoformat()
        last_success_at = config.get("last_success_at")
        last_error = (error or "Circuit breaker open")[:500]
    else:
        failures += 1
        circuit_open_until = None
        last_success_at = config.get("last_success_at")
        last_error = (error or status)[:500]
        if failures >= UPSTREAM_FAILURE_THRESHOLD:
            status = "circuit_open"
            circuit_open_until = (
                datetime.now(timezone.utc)
                + timedelta(seconds=UPSTREAM_CIRCUIT_OPEN_SECONDS)
            ).isoformat()

    UPSTREAM_FAILURES[name] = failures
    config.update({
        "connection_status": status,
        "consecutive_failures": failures,
        "last_checked_at": now,
        "last_success_at": last_success_at,
        "last_error": last_error,
        "circuit_open_until": circuit_open_until,
    })
    if discovery_succeeded:
        config["last_discovery_at"] = now

    with connect_db() as conn:
        conn.execute(
            """
            UPDATE upstreams
            SET connection_status = ?,
                consecutive_failures = ?,
                last_checked_at = ?,
                last_success_at = ?,
                last_discovery_at = ?,
                last_error = ?,
                circuit_open_until = ?
            WHERE name = ?
            """,
            (
                status,
                failures,
                now,
                last_success_at,
                config.get("last_discovery_at"),
                last_error,
                circuit_open_until,
                name,
            ),
        )


def record_upstream_success(
    server_name: str,
    discovery_succeeded: bool = False,
):
    update_upstream_status(
        server_name,
        "connected",
        discovery_succeeded=discovery_succeeded,
    )


def record_upstream_failure(server_name: str, exc: Exception):
    update_upstream_status(
        server_name,
        "error",
        error=f"{type(exc).__name__}: {str(exc)}",
    )


async def discover_upstream_tools(server_name: str, url: str):
    """Discover tools from one upstream using its configured auth and timeout."""
    config = get_upstream_config(server_name)
    headers = get_upstream_headers(config)
    timeout_seconds = min(
        UPSTREAM_DISCOVERY_TIMEOUT_SECONDS,
        int(config.get("timeout_seconds") or UPSTREAM_DISCOVERY_TIMEOUT_SECONDS),
    )

    async with streamablehttp_client(
        url,
        headers=headers,
        timeout=timedelta(seconds=UPSTREAM_CONNECT_TIMEOUT_SECONDS),
        sse_read_timeout=timedelta(seconds=timeout_seconds),
    ) as (read_stream, write_stream, _):
        async with ClientSession(read_stream, write_stream) as session:
            await asyncio.wait_for(
                session.initialize(),
                timeout=timeout_seconds,
            )
            result = await asyncio.wait_for(
                session.list_tools(),
                timeout=timeout_seconds,
            )
            return result.tools


async def call_upstream(
    server_name: str,
    tool_name: str,
    arguments: dict[str, Any],
):
    """Call an upstream MCP tool with authentication, timeout and breaker."""
    config = get_upstream_config(server_name)

    if circuit_is_open(config):
        raise RuntimeError(
            f"Upstream '{server_name}' circuit breaker is open until "
            f"{config.get('circuit_open_until')}"
        )

    url = config["url"]
    headers = get_upstream_headers(config)
    timeout_seconds = min(
        UPSTREAM_TOOL_TIMEOUT_SECONDS,
        int(config.get("timeout_seconds") or UPSTREAM_TOOL_TIMEOUT_SECONDS),
    )

    try:
        async with streamablehttp_client(
            url,
            headers=headers,
            timeout=UPSTREAM_CONNECT_TIMEOUT_SECONDS,
            sse_read_timeout=timeout_seconds,
        ) as (read_stream, write_stream, _):
            async with ClientSession(read_stream, write_stream) as session:
                await asyncio.wait_for(
                    session.initialize(),
                    timeout=timeout_seconds,
                )
                result = await asyncio.wait_for(
                    session.call_tool(tool_name, arguments),
                    timeout=timeout_seconds,
                )

        if result.isError:
            record_upstream_failure(
                server_name,
                RuntimeError("Upstream tool returned an error"),
            )
        else:
            record_upstream_success(server_name)

        return {
            "isError": result.isError,
            "content": [
                item.model_dump(mode="json")
                for item in result.content
            ],
        }
    except Exception as exc:
        record_upstream_failure(server_name, exc)
        raise

def safe_tool_name(name: str) -> str:
    """Convert a name into a predictable identifier."""
    normalized = re.sub(r"[^a-zA-Z0-9_]", "_", name)
    return normalized.strip("_") or "unnamed_tool"


def make_proxy(
    server_name: str,
    upstream_tool_name: str,
    input_schema: dict,
):
    """Create a dynamically typed proxy for an upstream tool."""
    properties = input_schema.get("properties", {})
    required_fields = set(input_schema.get("required", []))

    parameters = []

    for name, details in properties.items():
        schema_type = details.get("type", "string")

        type_mapping = {
            "string": str,
            "integer": int,
            "number": float,
            "boolean": bool,
            "object": dict,
            "array": list,
        }

        annotation = type_mapping.get(schema_type, Any)

        if name in required_fields:
            default = inspect.Parameter.empty
        else:
            if "default" in details:
                default = details["default"]
            elif schema_type == "integer":
                default = 10 if name == "per_page" else None
            else:
                default = None

            if default is None:
                annotation = Any

        parameters.append(
            inspect.Parameter(
                name=name,
                kind=inspect.Parameter.KEYWORD_ONLY,
                default=default,
                annotation=annotation,
            )
        )

    async def proxy(**arguments):
        arguments = {
            key: value
            for key, value in arguments.items()
            if value is not None
        }

        audit_started_at = datetime.now(timezone.utc).isoformat()
        started = time.perf_counter()
        outcome = "success"
        error_message = None

        user_id = current_user_id.get()
        username = current_username.get()

        try:
            user_role = current_user_role.get()

            write_tools = {
                ("notes", "create_note"),
                
                ("woocommerce", "update_order_status"),
            }

            if (
                (server_name, upstream_tool_name) in write_tools
                and user_role != "admin"
            ):
                raise PermissionError(
                    "Admin role required to execute this write tool"
                )

            result = await call_upstream(
                server_name,
                upstream_tool_name,
                arguments,
            )

            if result.get("isError"):
                outcome = "error"
                error_message = "Upstream tool returned an error"

            return result

        except Exception as exc:
            outcome = (
                "timeout"
                if isinstance(exc, (asyncio.TimeoutError, TimeoutError))
                else "error"
            )
            error_message = str(exc)[:500] or type(exc).__name__
            raise

        finally:
            duration_ms = int(
                (time.perf_counter() - started) * 1000
            )

            try:
                write_audit(
                    user_id=user_id,
                    username=username,
                    server_name=server_name,
                    tool_name=upstream_tool_name,
                    duration_ms=duration_ms,
                    outcome=outcome,
                    error_message=error_message,
                    started_at=audit_started_at,
                )
            except Exception:
                logger.exception(
                    "Failed to write MCP tool audit record"
                )

    proxy.__name__ = f"{server_name}_{upstream_tool_name}"
    proxy.__doc__ = f"Proxy for {server_name}.{upstream_tool_name}"
    proxy.__signature__ = inspect.Signature(parameters)

    return proxy


async def refresh_upstream_tools() -> dict[str, int]:
    """
    Discover enabled upstreams concurrently.

    Temporary failures do not remove previously registered tools. Disabled
    and deleted upstreams have their tools removed from the gateway.
    """
    candidates = [
        (name, config["url"])
        for name, config in list(UPSTREAM_CONFIG.items())
        if config.get("enabled", False)
    ]

    async def discover_one(name: str, url: str):
        config = UPSTREAM_CONFIG.get(name, {})
        if circuit_is_open(config):
            # Do not count a skipped probe as another failure.
            return name, [], None, False

        try:
            tools = await discover_upstream_tools(name, url)
            record_upstream_success(name, discovery_succeeded=True)
            return name, tools, None, True
        except Exception as exc:
            record_upstream_failure(name, exc)
            logger.warning(
                "Discovery failed for upstream '%s': %s",
                name,
                str(exc)[:300],
            )
            return name, [], exc, False

    results = await asyncio.gather(
        *(discover_one(name, url) for name, url in candidates),
        return_exceptions=False,
    )

    discovered: dict[str, list[Any]] = {}
    successful_names: set[str] = set()

    for name, tools, error, succeeded in results:
        if succeeded:
            discovered[name] = tools
            successful_names.add(name)

    registered_count = 0

    async with REGISTRATION_LOCK:
        for server_name, tools in discovered.items():
            new_routes: dict[str, dict[str, str]] = {}

            for upstream_tool in tools:
                gateway_name = (
                    f"{safe_tool_name(server_name)}__"
                    f"{safe_tool_name(upstream_tool.name)}"
                )
                route = {
                    "server": server_name,
                    "tool": upstream_tool.name,
                }
                existing = TOOL_ROUTES.get(gateway_name)

                if existing and existing != route:
                    logger.error(
                        "Tool name collision for '%s'",
                        gateway_name,
                    )
                    continue

                if existing == route:
                    new_routes[gateway_name] = route
                    registered_count += 1
                    continue

                try:
                    proxy = make_proxy(
                        server_name,
                        upstream_tool.name,
                        upstream_tool.inputSchema,
                    )
                    gateway_mcp.add_tool(
                        proxy,
                        name=gateway_name,
                        description=(
                            upstream_tool.description
                            or f"Proxy for {upstream_tool.name}"
                        ),
                    )
                    new_routes[gateway_name] = route
                    registered_count += 1
                except Exception:
                    logger.exception(
                        "Could not register tool '%s'",
                        gateway_name,
                    )

            # Remove stale tools only after successful discovery.
            old_names = [
                name
                for name, route in TOOL_ROUTES.items()
                if route["server"] == server_name
                and name not in new_routes
            ]
            for old_name in old_names:
                gateway_mcp.remove_tool(old_name)
                TOOL_ROUTES.pop(old_name, None)

            TOOL_ROUTES.update(new_routes)

        # Remove tools belonging to disabled or deleted upstreams.
        inactive_names = [
            name
            for name, route in TOOL_ROUTES.items()
            if route["server"] not in UPSTREAM_CONFIG
            or not UPSTREAM_ENABLED.get(route["server"], False)
        ]
        for name in inactive_names:
            gateway_mcp.remove_tool(name)
            TOOL_ROUTES.pop(name, None)

    return {
        "upstreams_discovered": len(successful_names),
        "tools_registered": registered_count,
    }


async def upstream_monitor_loop():
    """Periodically refresh upstream health and tool catalogues."""
    while True:
        await asyncio.sleep(UPSTREAM_HEALTH_CHECK_INTERVAL_SECONDS)
        try:
            await refresh_upstream_tools()
        except asyncio.CancelledError:
            raise
        except Exception:
            logger.exception("Background upstream refresh failed")

def load_saved_upstreams():
    """Load persisted upstreams and decrypt credentials for runtime use."""
    saved = load_upstreams()

    if not saved:
        for name, url in DEFAULT_UPSTREAMS.items():
            save_upstream(
                name=name,
                url=url,
                enabled=True,
                auth_mode="none",
            )

    saved = load_upstreams(include_secrets=True)

    UPSTREAMS.clear()
    UPSTREAM_ENABLED.clear()
    UPSTREAM_CONFIG.clear()
    UPSTREAM_FAILURES.clear()

    for upstream in saved:
        name = upstream["name"]
        UPSTREAMS[name] = upstream["url"]
        UPSTREAM_ENABLED[name] = upstream["enabled"]
        UPSTREAM_CONFIG[name] = dict(upstream)
        UPSTREAM_FAILURES[name] = upstream.get(
            "consecutive_failures", 0
        )

    logger.info("Loaded %d upstream configurations", len(saved))

@asynccontextmanager
async def lifespan(app: FastAPI):
    init_db()
    load_saved_upstreams()

    # Gateway startup must not fail merely because an upstream is down.
    try:
        await refresh_upstream_tools()
    except Exception:
        logger.exception("Initial upstream discovery failed")

    monitor_task = asyncio.create_task(
        upstream_monitor_loop(),
        name="upstream-monitor",
    )
    try:
        async with gateway_mcp.session_manager.run():
            yield
    finally:
        monitor_task.cancel()
        try:
            await monitor_task
        except asyncio.CancelledError:
            pass

@app.post("/admin/users", status_code=201)
async def admin_create_user(payload: UserCreate):
    try:
        user_id = create_user(
            payload.username,
            payload.role,
            payload.email,
        )
    except Exception as exc:
        if "UNIQUE constraint failed" in str(exc):
            raise HTTPException(
                status_code=409,
                detail="Username or email already exists",
            ) from exc

        if isinstance(exc, ValueError):
            raise HTTPException(
                status_code=400,
                detail=str(exc),
            ) from exc

        raise

    return {
        "user_id": user_id,
        "username": payload.username,
        "role": payload.role,
    }


@app.get("/admin/users")
async def admin_list_users():
    return {"users": list_users()}


@app.patch("/admin/users/{user_id}")
async def admin_update_user(user_id: str, payload: UserRoleUpdate):
    if user_id == current_user_id.get() and payload.role != "admin":
        raise HTTPException(
            status_code=400, detail="You cannot demote yourself"
        )

    try:
        updated = update_user_role(user_id, payload.role)
    except ValueError as exc:
        raise HTTPException(status_code=400, detail=str(exc)) from exc

    if not updated:
        raise HTTPException(status_code=404, detail="User not found")

    return {"user_id": user_id, "role": payload.role}


@app.delete("/admin/users/{user_id}")
async def admin_delete_user(user_id: str):
    if user_id == current_user_id.get():
        raise HTTPException(
            status_code=400, detail="You cannot remove your own account"
        )

    try:
        deleted = delete_user(user_id)
    except ValueError as exc:
        raise HTTPException(status_code=400, detail=str(exc)) from exc

    if not deleted:
        raise HTTPException(status_code=404, detail="User not found")

    return {"message": "User removed", "user_id": user_id}

@app.get("/admin/keys")
async def admin_list_keys():
    with connect_db() as conn:
        rows = conn.execute(
            """
            SELECT
                k.id AS key_id,
                k.user_id,
                u.username,
                k.label,
                k.key_prefix,
                k.last_used_at,
                k.created_at,
                k.revoked_at,
                k.revoked
            FROM api_keys AS k
            JOIN users AS u ON u.id = k.user_id
            ORDER BY k.id DESC
            LIMIT 500
            """
        ).fetchall()

    return {"keys": [dict(row) for row in rows]}

@app.get("/api/tools")
async def list_my_tools():
    """List tools from enabled upstreams for any authenticated API user."""
    registered = await gateway_mcp.list_tools()

    tools = []
    for tool in registered:
        route = TOOL_ROUTES.get(tool.name)
        if route is None:
            continue

        server_name = route["server"]
        config = UPSTREAM_CONFIG.get(server_name)

        # Do not expose tools from disabled, deleted, or unknown upstreams.
        if not config or not config.get("enabled", False):
            continue

        tools.append({
            "gateway_name": tool.name,
            "upstream_server": server_name,
            "upstream_tool": route["tool"],
            "description": tool.description,
            "input_schema": tool.inputSchema,
        })

    tools.sort(key=lambda item: (
        item["upstream_server"],
        item["upstream_tool"],
    ))
    return {"tools": tools}

                username,
                server_name,
                tool_name,
                started_at,
                timestamp,
                duration_ms,
                outcome,
                error_message
            FROM audit_logs
            {where_clause}
            ORDER BY id DESC
            LIMIT ? OFFSET ?
            """,

@app.patch("/admin/upstreams/{name}")
async def update_upstream(name: str, payload: UpstreamUpdate):
    current = UPSTREAM_CONFIG.get(name)
    if current is None:
        raise HTTPException(status_code=404, detail="Upstream not found")

    new_url = (
        validate_upstream_url(payload.url)
        if payload.url is not None
        else current["url"]
    )
    new_enabled = (
        payload.enabled
        if payload.enabled is not None
        else current["enabled"]
    )
    new_auth_mode = (
        payload.auth_mode
        if payload.auth_mode is not None
        else current.get("auth_mode", "none")
    )
    new_header_name = (
        payload.custom_header_name
        if payload.custom_header_name is not None
        else current.get("custom_header_name")
    )
    new_timeout = (
        payload.timeout_seconds
        if payload.timeout_seconds is not None
        else current.get("timeout_seconds", 20)
    )

    try:
        save_upstream(
            name=name,
            url=new_url,
            enabled=new_enabled,
            auth_mode=new_auth_mode,
            credential=payload.credential,
            custom_header_name=new_header_name,
            timeout_seconds=new_timeout,
        )
    except ValueError as exc:
        raise HTTPException(status_code=400, detail=str(exc)) from exc

    load_saved_upstreams()

    if not new_enabled:
        update_upstream_status(name, "disabled")

    result = await refresh_upstream_tools()
    updated = next(
        item for item in load_upstreams() if item["name"] == name
    )

    return {
        "message": "Upstream updated",
        **updated,
        **result,
    }

@app.post("/api/chat")
async def chat_with_agent(payload: ChatRequest, request: Request):
    """Chat with an LLM that can invoke tools through this MCP gateway."""
    import json
    import os
    import httpx
    from mcp import ClientSession
    from mcp.client.streamable_http import streamable_http_client

    
    api_key = request.headers.get("x-api-key")

    llm_key = os.getenv("LLM_API_KEY") or os.getenv("OPENAI_API_KEY")
    base_url = os.getenv("LLM_BASE_URL")
    model = os.getenv("LLM_MODEL")
    gateway_url = os.getenv(
        "GATEWAY_MCP_URL",
        "http://127.0.0.1:8081/mcp/mcp",
    )

    if not api_key:
        raise HTTPException(
            status_code=401,
            detail="Missing gateway API key.",
        )

    if not llm_key:
        raise HTTPException(
            status_code=503,
            detail="Chat model is not configured. Set LLM_API_KEY.",
        )

    if not base_url:
        raise HTTPException(
            status_code=503,
            detail="LLM_BASE_URL is not configured.",
        )

    if not model:
        raise HTTPException(
            status_code=503,
            detail="LLM_MODEL is not configured.",
        )

    if not payload.message.strip():
        raise HTTPException(
            status_code=422,
            detail="Message cannot be empty.",
        )

    base_url = base_url.rstrip("/")

    messages = [{
        "role": "system",
        "content": (
            "You are a helpful team assistant. Use the available MCP tools "
            "when needed to answer questions or perform requested actions. "
            "Never claim an action succeeded unless a tool result confirms it. "
            "Ask for clarification when required arguments are missing."
        )
    }]

    # Accept only ordinary conversation history, not client-supplied tool results.
    for item in (payload.history or [])[-12:]:
        role = item.get("role")
        content = item.get("content")
        if role in ("user", "assistant") and isinstance(content, str):
            messages.append({"role": role, "content": content})

    messages.append({"role": "user", "content": payload.message})
    executed_calls = []

    mcp_http_client = httpx.AsyncClient(
        headers={"x-api-key": api_key},
        timeout=60.0,
    )
    try:
        async with streamable_http_client(
            gateway_url,
            http_client=mcp_http_client,
        ) as (read_stream, write_stream, _):
            async with ClientSession(read_stream, write_stream) as session:
                await session.initialize()
                listed = await session.list_tools()

                llm_tools = []
                for tool in listed.tools:
                    llm_tools.append({
                        "type": "function",
                        "function": {
                            "name": tool.name,
                            "description": tool.description or tool.name,
                            "parameters": tool.inputSchema or {
                                "type": "object", "properties": {}
                            },
                        },
                    })

                async with httpx.AsyncClient(timeout=60.0) as client:
                    for _ in range(6):
                        body = {
                            "model": model,
                            "messages": messages,
                        }
                        if llm_tools:
                            body["tools"] = llm_tools
                            body["tool_choice"] = "auto"

                        response = await client.post(
                            f"{base_url}/chat/completions",
                            headers={"Authorization": f"Bearer {llm_key}"},
                            json=body,
                        )
                        response.raise_for_status()
                        choice = response.json()["choices"][0]
                        assistant_message = choice["message"]
                        calls = assistant_message.get("tool_calls") or []

                        if not calls:
                            return {
                                "answer": assistant_message.get("content") or "",
                                "tool_calls": executed_calls,
                                "model": model,
                            }

                        messages.append(assistant_message)

                        for call in calls:
                            function = call["function"]
                            tool_name = function["name"]

                            try:
                                arguments = json.loads(
                                    function.get("arguments") or "{}"
                                )
                                result = await session.call_tool(
                                    tool_name, arguments=arguments
                                )

                                content = []
                                for item in (result.content or []):
                                    text_value = getattr(item, "text", None)
                                    if text_value is not None:
                                        content.append(text_value)
                                    elif hasattr(item, "model_dump"):
                                        content.append(
                                            item.model_dump(mode="json")
                                        )
                                    else:
                                        content.append(str(item))

                                tool_output = json.dumps({
                                    "isError": bool(result.isError),
                                    "content": content,
                                }, ensure_ascii=False)

                                executed_calls.append({
                                    "tool": tool_name,
                                    "arguments": arguments,
                                    "result": tool_output[:4000],
                                    "isError": bool(result.isError),
                                })
                            except Exception as exc:
                                tool_output = json.dumps({
                                    "error": str(exc)[:1000]
                                })
                                executed_calls.append({
                                    "tool": tool_name,
                                    "error": str(exc)[:1000],
                                })

                            messages.append({
                                "role": "tool",
                                "tool_call_id": call["id"],
                                "content": tool_output,
                            })

                return {
                    "answer": "The tool-call limit was reached. Please refine your request.",
                    "tool_calls": executed_calls,
                    "model": model,
                }

    except HTTPException:
        raise
    except httpx.HTTPStatusError as exc:
        raise HTTPException(
            status_code=502,
            detail=f"Chat provider returned HTTP {exc.response.status_code}."
        ) from exc
    except httpx.TimeoutException as exc:
        raise HTTPException(
            status_code=504,
            detail="The chat provider timed out."
        ) from exc
    except Exception as exc:
        logger.exception("Chat agent failed with an unexpected exception")
        raise HTTPException(
            status_code=502,
            detail=f"Chat agent failed: {type(exc).__name__}. Check gateway logs."
        ) from exc
    finally:
        await mcp_http_client.aclose()

def validate_configuration() -> None:
    """Validate required configuration without exposing credentials."""
    if not STORE_URL:
        raise RuntimeError("WOOCOMMERCE_URL is not configured.")

    parsed = urlsplit(STORE_URL)

    if parsed.scheme not in {"http", "https"} or not parsed.netloc:
        raise RuntimeError(
            "WOOCOMMERCE_URL must be a valid HTTP(S) URL."
        )

    if parsed.username or parsed.password or parsed.query or parsed.fragment:
        raise RuntimeError(
            "WOOCOMMERCE_URL must not contain credentials, "
            "query parameters, or fragments."
        )

    if not CONSUMER_KEY or not CONSUMER_SECRET:
        raise RuntimeError(
            "WooCommerce API credentials are not configured."
        )

    if REQUEST_TIMEOUT <= 0:
        raise RuntimeError("WOOCOMMERCE_TIMEOUT must be greater than zero.")

    try:
        ZoneInfo(STORE_TIMEZONE)
    except (ZoneInfoNotFoundError, ValueError) as exc:
        raise RuntimeError(
            "STORE_TIMEZONE must be a valid IANA timezone, "
            "such as Asia/Kolkata."
        ) from exc


def api_base_url() -> str:
    """Build the WooCommerce REST API base URL."""
    validate_configuration()
    return f"{STORE_URL}/wp-json/wc/v3"


def _pct(value: str) -> str:
    """OAuth percent encoding."""
    return quote(str(value), safe="~-._")


def _positive_price(value: str) -> str:
    """Validate and normalize a non-negative product price."""
    try:
        price = Decimal(value)
    except (InvalidOperation, TypeError, ValueError):
        raise ValueError(
            "regular_price must be a valid decimal number."
        ) from None

    if not price.is_finite() or price < 0:
        raise ValueError(
            "regular_price must be a finite, non-negative number."
        )

    return format(price, "f")


def _parse_datetime(value: str, field_name: str) -> datetime:
    """Parse an ISO-8601 datetime that includes a timezone."""
    try:
        parsed = datetime.fromisoformat(
            value.strip().replace("Z", "+00:00")
        )
    except ValueError:
        raise ValueError(
            f"{field_name} must be a valid ISO-8601 datetime."
        ) from None

    if parsed.tzinfo is None or parsed.utcoffset() is None:
        raise ValueError(
            f"{field_name} must include a timezone offset."
        )

    return parsed

def _oauth_params(method: str, url: str, params: dict) -> dict:
    """Sign a WooCommerce REST request using HMAC-SHA256."""
    if not CONSUMER_KEY or not CONSUMER_SECRET:
        raise RuntimeError(
            "WooCommerce API credentials are not configured."
        )

    oauth = {
        "oauth_consumer_key": CONSUMER_KEY,
        "oauth_nonce": secrets.token_hex(16),
        "oauth_signature_method": "HMAC-SHA256",
        "oauth_timestamp": str(int(time.time())),
        "oauth_version": "1.0",
    }

    all_params = [
        (str(key), str(value))
        for key, value in {**params, **oauth}.items()
        if value is not None
    ]

    encoded_params = sorted(
        (_pct(key), _pct(value))
        for key, value in all_params
    )
    normalized = "&".join(
        f"{key}={value}" for key, value in encoded_params
    )

    parsed = urlsplit(url)
    base_url = urlunsplit(
        (parsed.scheme, parsed.netloc, parsed.path, "", "")
    )

    base_string = "&".join(
        [
            method.upper(),
            _pct(base_url),
            _pct(normalized),
        ]
    )

    signing_key = f"{_pct(CONSUMER_SECRET)}&"
    digest = hmac.new(
        signing_key.encode(),
        base_string.encode(),
        hashlib.sha256,
    ).digest()

    oauth["oauth_signature"] = base64.b64encode(digest).decode()
    return oauth


async def wc_request(
    method: str,
    endpoint: str,
    *,
    params: dict | None = None,
    payload: dict | None = None,
):
    """Call WooCommerce and return parsed JSON."""
    url = f"{api_base_url()}/{endpoint.lstrip('/')}"
    query = {
        key: value
        for key, value in (params or {}).items()
        if value is not None
    }
    query.update(_oauth_params(method, url, query))

    try:
        async with httpx.AsyncClient(
            timeout=REQUEST_TIMEOUT
        ) as client:
            response = await client.request(
                method,
                url,
                params=query,
                json=payload,
            )
    except httpx.TimeoutException:
        raise RuntimeError(
            f"WooCommerce API request timed out after "
            f"{REQUEST_TIMEOUT:g} seconds."
        ) from None
    except httpx.RequestError as exc:
        # Do not include the request URL, query parameters, or credentials.
        raise RuntimeError(
            f"Could not connect to WooCommerce "
            f"({type(exc).__name__})."
        ) from None

    if not response.is_success:
        message = (
            f"WooCommerce API returned HTTP {response.status_code}."
        )

        try:
            body = response.json()
            if isinstance(body, dict):
                api_message = body.get("message")
                api_code = body.get("code")

                if isinstance(api_code, str):
                    message += f" Error code: {api_code}."
                if isinstance(api_message, str):
                    message += f" {api_message[:300]}"
        except ValueError:
            message += " The response was not valid JSON."

        raise RuntimeError(message)

    try:
        return response.json()
    except ValueError:
        raise RuntimeError(
            "WooCommerce API returned invalid JSON."
        ) from None

@mcp.tool()
async def list_orders(
    after: str = "",
    before: str = "",
    per_page: int = 20,
    today: bool = False,
) -> list[dict]:
    """
    List orders by date.

    Set today=True to filter from midnight to midnight in STORE_TIMEZONE.
    Otherwise, optionally provide timezone-aware ISO-8601 after/before values.
    """
    if not 1 <= per_page <= 100:
        raise ValueError("per_page must be between 1 and 100.")

    after = after.strip()
    before = before.strip()

    if today and (after or before):
        raise ValueError(
            "Use today=True or provide after/before, not both."
        )

    query = {
        "per_page": per_page,
        "orderby": "date",
        "order": "desc",
    }

    if today:
        try:
            store_tz = ZoneInfo(STORE_TIMEZONE)
        except (ZoneInfoNotFoundError, ValueError):
            raise RuntimeError(
                "STORE_TIMEZONE is not a valid timezone."
            ) from None

        now_local = datetime.now(store_tz)
        start_local = now_local.replace(
            hour=0, minute=0, second=0, microsecond=0
        )
        end_local = start_local + timedelta(days=1)

        # Convert local calendar-day boundaries to UTC instants.
        # This also handles daylight-saving transitions correctly.
        start_utc = start_local.astimezone(timezone.utc)
        end_utc = end_local.astimezone(timezone.utc)

        query["after"] = start_utc.isoformat()
        query["before"] = end_utc.isoformat()

    else:
        if after:
            query["after"] = _parse_datetime(after, "after").isoformat()

        if before:
            query["before"] = _parse_datetime(before, "before").isoformat()

    result = await wc_request(
        "GET",
        "orders",
        params=query,
    )

    if not isinstance(result, list):
        raise RuntimeError("Unexpected response while listing orders.")

    return [
        {
            "id": order.get("id"),
            "number": order.get("number"),
            "status": order.get("status"),
            "date_created": order.get("date_created"),
            "total": order.get("total"),
            "currency": order.get("currency"),
            "payment_method_title": order.get("payment_method_title"),
            "line_items": [
                {
                    "product_id": item.get("product_id"),
                    "name": item.get("name"),
                    "quantity": item.get("quantity"),
                    "total": item.get("total"),
                }
                for item in order.get("line_items", [])
            ],
        }
        for order in result
    ]

function extractMessage(body: string, fallback: string): string {
  if (!body) return fallback

  try {
    const parsed = JSON.parse(body)
    const detail = parsed?.detail

    if (typeof detail === 'string') return detail

    // FastAPI/Pydantic validation errors (422)
    if (Array.isArray(detail)) {
      return detail
        .map((item) => {
          const where = Array.isArray(item?.loc)
            ? item.loc.filter((part: unknown) => part !== 'body').join('.')
            : ''
          return where ? `${where}: ${item?.msg}` : String(item?.msg ?? item)
        })
        .join('; ')
    }

    if (typeof parsed?.message === 'string') return parsed.message
  } catch {
    // Not JSON; fall through to the raw text.
  }

  return body.slice(0, 300)
}

export async function apiRequest<T>(
  path: string,
  options: RequestInit = {},
): Promise<T> {
  const headers = new Headers(options.headers)

  headers.set('Accept', 'application/json')

  if (options.body && !headers.has('Content-Type')) {
    headers.set('Content-Type', 'application/json')
  }

  // Remove stale credentials before applying the active credential.
  headers.delete('X-Admin-Token')
  headers.delete('X-API-Key')
  headers.delete('Authorization')

  const credential = authConfig.credential.trim()
  const isAdminEndpoint = path === '/admin' || path.startsWith('/admin/')
  const isHealthEndpoint = path === '/health'

  if (credential && !isHealthEndpoint) {
    if (authConfig.mode === 'admin-token' && isAdminEndpoint) {
      headers.set('X-Admin-Token', credential)
    } else if (authConfig.mode === 'api-key') {
      headers.set('X-API-Key', credential)
    }
  }

  let response: Response

  try {
    response = await fetch(`${API_BASE_URL}${path}`, {
      ...options,
      headers,
    })
  } catch {
    throw new ApiError(
      0,
      `Cannot reach the gateway${API_BASE_URL ? ` at ${API_BASE_URL}` : ''}.`,
    )
  }

  if (!response.ok) {
    const body = await response.text()
    const message = extractMessage(body, response.statusText)

    if (
      response.status === 401 &&
      credential &&
      path !== '/api/me' &&
      unauthorizedHandler
    ) {
      unauthorizedHandler()
    }

    throw new ApiError(response.status, message)
  }

  if (response.status === 204) {
    return undefined as T
  }

  return response.json() as Promise<T>
}
